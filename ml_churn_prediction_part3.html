from flask import Flask, request, jsonify
import pickle

# Initialize Flask app

# Load model artifacts at startup

    model = pickle.load(f)

with open('model_metadata.pkl', 'rb') as f:
    metadata = pickle.load(f)

feature_names = metadata['feature_names']
print(f"✅ Model loaded: {metadata['model_name']}")
print(f"✅ Training date: {metadata['training_date']}")
print(f"✅ Expected features: {len(feature_names)}")

def preprocess_input(data):
    """Preprocess input data to match training format"""
    try:
        # Convert to DataFrame
        df = pd.DataFrame([data])

        # Encode categorical variables (same as training)
        categorical_cols = ['contract_type', 'internet_service', 'online_security',
                          'tech_support', 'payment_method', 'paperless_billing']

        # One-hot encoding
        df_encoded = pd.get_dummies(df, columns=categorical_cols, drop_first=True)

        # Ensure all expected features are present
        for feature in feature_names:
            if feature not in df_encoded.columns:
                df_encoded[feature] = 0

        # Ensure same order as training
        df_encoded = df_encoded[feature_names]

        return df_encoded

    except Exception as e:
        logger.error(f"Preprocessing error: {str(e)}")
        raise

@app.route('/health', methods=['GET'])
def health_check():
    """Health check endpoint"""
    return jsonify({
        'status': 'healthy',
        'model': metadata['model_name'],
        'timestamp': datetime.now().isoformat()
    }), 200

@app.route('/predict', methods=['POST'])
def predict():
    """Predict churn for a single customer"""
    try:
        # Get input data
        data = request.get_json()

        if not data:
            return jsonify({'error': 'No input data provided'}), 400

        # Preprocess
        X = preprocess_input(data)

        # Make prediction
        prediction = model.predict(X)[0]
        probability = model.predict_proba(X)[0, 1]

        # Determine risk level
        if probability >= 0.7:
            risk_level = 'HIGH'
            recommendation = 'Immediate retention intervention recommended'
        elif probability >= 0.5:
            risk_level = 'MEDIUM'
            recommendation = 'Monitor closely and consider retention offer'
        else:
            risk_level = 'LOW'
            recommendation = 'Continue standard service'

        # Log prediction
        logger.info(f"Prediction: {prediction}, Probability: {probability:.4f}, Risk: {risk_level}")

        # Return response
        response = {
            'customer_id': data.get('customer_id', 'unknown'),
            'prediction': 'WILL CHURN' if prediction == 1 else 'WILL STAY',
            'churn_probability': round(float(probability), 4),
            'churn_percentage': round(float(probability) * 100, 2),
            'risk_level': risk_level,
            'recommendation': recommendation,
            'timestamp': datetime.now().isoformat()
        }

        return jsonify(response), 200

    except Exception as e:
        logger.error(f"Prediction error: {str(e)}")
        return jsonify({'error': str(e)}), 500

@app.route('/predict/batch', methods=['POST'])
def predict_batch():
    """Predict churn for multiple customers"""
    try:
        # Get input data
        data = request.get_json()

        if not data or 'customers' not in data:
            return jsonify({'error': 'No customer data provided'}), 400

        customers = data['customers']
        results = []

        for customer in customers:
            try:
                # Preprocess
                X = preprocess_input(customer)

                # Make prediction
                prediction = model.predict(X)[0]
                probability = model.predict_proba(X)[0, 1]

                # Determine risk level
                if probability >= 0.7:
                    risk_level = 'HIGH'
                elif probability >= 0.5:
                    risk_level = 'MEDIUM'
                else:
                    risk_level = 'LOW'

                results.append({
                    'customer_id': customer.get('customer_id', 'unknown'),
                    'prediction': 'WILL CHURN' if prediction == 1 else 'WILL STAY',
                    'churn_probability': round(float(probability), 4),
                    'risk_level': risk_level
                })

            except Exception as e:
                results.append({
                    'customer_id': customer.get('customer_id', 'unknown'),
                    'error': str(e)
                })

        return jsonify({
            'total_customers': len(customers),
            'predictions': results,
            'timestamp': datetime.now().isoformat()
        }), 200

    except Exception as e:
        logger.error(f"Batch prediction error: {str(e)}")
        return jsonify({'error': str(e)}), 500

⚙️ 3. Automated Batch Scoring Pipeline

For TeleConnect's 50,000 customers, we need an automated system that scores all customers weekly and identifies those at high risk of churn.

3Weekly Batch Scoring Script

Create a script that processes all customers and generates a risk report:

import pandas as pd
import pickle
import numpy as np
from datetime import datetime
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
import logging

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler(f'batch_scoring_{datetime.now().strftime("%Y%m%d")}.log'),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)

def load_model():
    """Load trained model and metadata"""
    logger.info("Loading model artifacts...")

    with open('churn_prediction_model.pkl', 'rb') as f:
        model = pickle.load(f)

    with open('model_metadata.pkl', 'rb') as f:
        metadata = pickle.load(f)

    logger.info(f"✅ Model loaded: {metadata['model_name']}")
    return model, metadata

def load_customer_data():
    """Load customer data from database/CSV"""
    logger.info("Loading customer data...")

    # In production, this would connect to your database
    # For this example, we'll use our CSV file
    df = pd.read_csv('teleconnect_customer_data.csv')

    logger.info(f"✅ Loaded {len(df)} customers")
    return df

def preprocess_data(df, feature_names):
    """Preprocess customer data for prediction"""
    logger.info("Preprocessing data...")

    # Separate features and target
    X = df.drop(['customer_id', 'churn'], axis=1, errors='ignore')
    customer_ids = df['customer_id']

    # Encode categorical variables
    categorical_cols = ['contract_type', 'internet_service', 'online_security',
                       'tech_support', 'payment_method', 'paperless_billing']

    X_encoded = pd.get_dummies(X, columns=categorical_cols, drop_first=True)

    # Ensure all expected features are present
    for feature in feature_names:
        if feature not in X_encoded.columns:
            X_encoded[feature] = 0

    # Ensure same order as training
    X_encoded = X_encoded[feature_names]

    logger.info("✅ Preprocessing complete")
    return X_encoded, customer_ids

def make_predictions(model, X):
    """Generate predictions for all customers"""
    logger.info("Making predictions...")

    predictions = model.predict(X)
    probabilities = model.predict_proba(X)[:, 1]

    logger.info("✅ Predictions complete")
    return predictions, probabilities

def generate_risk_report(customer_ids, predictions, probabilities, df_original):
    """Generate detailed risk report"""
    logger.info("Generating risk report...")

    # Create results DataFrame
    results = pd.DataFrame({
        'customer_id': customer_ids,
        'churn_prediction': predictions,
        'churn_probability': probabilities
    })

    # Add risk level
    results['risk_level'] = pd.cut(
        results['churn_probability'],
        bins=[0, 0.5, 0.7, 1.0],
        labels=['LOW', 'MEDIUM', 'HIGH']
    )

    # Merge with original data for context
    results = results.merge(
        df_original[['customer_id', 'tenure', 'monthly_charges', 'contract_type',
                     'payment_method', 'num_support_tickets']],
        on='customer_id',
        how='left'
    )

    # Sort by probability (highest risk first)
    results = results.sort_values('churn_probability', ascending=False)

    # Generate summary statistics
    total_customers = len(results)
    high_risk = len(results[results['risk_level'] == 'HIGH'])
    medium_risk = len(results[results['risk_level'] == 'MEDIUM'])
    low_risk = len(results[results['risk_level'] == 'LOW'])
    predicted_churners = len(results[results['churn_prediction'] == 1])

    summary = {
        'scoring_date': datetime.now().strftime('%Y-%m-%d %H:%M:%S'),
        'total_customers': total_customers,
        'high_risk_customers': high_risk,
        'medium_risk_customers': medium_risk,
        'low_risk_customers': low_risk,
        'predicted_churners': predicted_churners,
        'churn_rate_forecast': f"{(predicted_churners/total_customers)*100:.2f}%"
    }

    logger.info("✅ Risk report generated")
    return results, summary

def save_results(results, summary):
    """Save results to CSV and summary to JSON"""
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')

    # Save detailed results
    results_file = f'churn_predictions_{timestamp}.csv'
    results.to_csv(results_file, index=False)
    logger.info(f"✅ Results saved to: {results_file}")

    # Save high-risk customers separately
    high_risk = results[results['risk_level'] == 'HIGH']
    high_risk_file = f'high_risk_customers_{timestamp}.csv'
    high_risk.to_csv(high_risk_file, index=False)
    logger.info(f"✅ High-risk customers saved to: {high_risk_file}")

    # Print summary
    print("\n" + "=" * 80)
    print("BATCH SCORING SUMMARY")
    print("=" * 80)
    for key, value in summary.items():
        print(f"{key.replace('_', ' ').title()}: {value}")
    print("=" * 80 + "\n")

    return results_file, high_risk_file

def main():
    """Main batch scoring pipeline"""
    try:
        logger.info("=" * 80)
        logger.info("STARTING WEEKLY BATCH SCORING PIPELINE")
        logger.info("=" * 80)

        # Step 1: Load model
        model, metadata = load_model()

        # Step 2: Load customer data
        df = load_customer_data()

        # Step 3: Preprocess data
        X, customer_ids = preprocess_data(df, metadata['feature_names'])

        # Step 4: Make predictions
        predictions, probabilities = make_predictions(model, X)

        # Step 5: Generate risk report
        results, summary = generate_risk_report(customer_ids, predictions, probabilities, df)

        # Step 6: Save results
        results_file, high_risk_file = save_results(results, summary)

        # Step 7: Send alert
        send_email_alert(summary, high_risk_file)

        logger.info("=" * 80)
        logger.info("✅ BATCH SCORING PIPELINE COMPLETED SUCCESSFULLY")
        logger.info("=" * 80)

    except Exception as e:
        logger.error(f"❌ Pipeline failed: {str(e)}")
        raise

if __name__ == '__main__':
    main()

import pandas as pd
import numpy as np
from datetime import datetime, timedelta

import warnings

class ModelMonitor:

        predictions_df['prediction_date'] = datetime.now()
        predictions_df.to_csv(

            mode='a',

            index=False
        )
        print(f"✅ Logged {len(predictions_df)} predictions")

    def evaluate_performance(self, days_back=30):
        """Evaluate model performance on recent actual outcomes"""
        print(f"\n{'='*80}")
        print(f"MODEL PERFORMANCE EVALUATION - Last {days_back} Days")
        print(f"{'='*80}\n")

        # In production, you would query actual churn outcomes from your database
        # For this example, we'll simulate with our dataset

        # Load original data with actual outcomes
        actual_df = pd.read_csv('teleconnect_customer_data.csv')

            predictions_df = pd.read_csv(f'predictions_log_{datetime.now().strftime("%Y%m%d")}.csv')

            print("⚠️ No prediction log found")
            return

        # Merge predictions with actual outcomes
        eval_df = predictions_df.merge(
            actual_df[['customer_id', 'churn']],
            on='customer_id',
            how='inner'
        )

        if len(eval_df) == 0:
            print("⚠️ No matching data for evaluation")
            return

        # Calculate metrics
        y_true = eval_df['churn']
        y_pred = eval_df['churn_prediction']
        y_proba = eval_df['churn_probability']

        accuracy = accuracy_score(y_true, y_pred)
        precision = precision_score(y_true, y_pred)
        recall = recall_score(y_true, y_pred)
        roc_auc = roc_auc_score(y_true, y_proba)

        print(f"  Precision: {precision:.4f} ({precision*100:.2f}%)")

        print(f"  ROC-AUC:   {roc_auc:.4f}")

        performance_record = {
            'date': datetime.now(),
            'accuracy': accuracy,
            'precision': precision,
            'recall': recall,
            'roc_auc': roc_auc,
            'samples': len(eval_df)
        }
        self.performance_log.append(performance_record)

        # Check for performance degradation
        if accuracy < 0.75 or roc_auc < 0.80:
            print("\n⚠️ WARNING: Model performance below threshold!")
            print("   Recommendation: Consider retraining the model")

import pandas as pd

class BusinessImpactTracker:
    """Track business impact of churn prediction system"""

    def __init__(self, baseline_churn_rate=0.25):

        self.monthly_revenue_per_customer = 100
        self.retention_cost = 100

    def calculate_roi(self, customers_targeted, customers_saved, months=12):
        """Calculate ROI from retention campaigns"""

        print(f"\n{'='*80}")
        print("BUSINESS IMPACT & ROI CALCULATION")
        print(f"{'='*80}\n")

        # Revenue saved from retained customers
        revenue_saved = customers_saved * self.monthly_revenue_per_customer * months

        # Acquisition costs avoided
        acquisition_saved = customers_saved * self.acquisition_cost

        # Total benefit
        total_benefit = revenue_saved + acquisition_saved

        # Total cost
        retention_campaign_cost = customers_targeted * self.retention_cost
        ml_system_cost = 50000  # One-time + annual maintenance
        total_cost = retention_campaign_cost + ml_system_cost

        # ROI calculation
        roi = ((total_benefit - total_cost) / total_cost) * 100

        print(f"Campaign Performance:")
        print(f"  Customers Targeted:        {customers_targeted:,}")
        print(f"  Customers Successfully Saved: {customers_saved:,}")

✅ 7. Production Best Practices

🔒 Security & Privacy

  • Data Protection: Encrypt customer data in transit and at rest
  • Access Control: Implement role-based access to predictions
  • Audit Logging: Track who accessed what predictions and when
  • GDPR Compliance: Allow customers to opt-out of ML-based decisions
  • Data Retention: Define policies for how long predictions are stored

⚡ Performance & Scalability

  • Caching: Cache model predictions to reduce computation
  • Batch Processing: Process customers in batches during off-peak hours
  • Load Balancing: Distribute API requests across multiple servers
  • Database Optimization: Index customer_id fields for fast lookups
  • Async Processing: Use message queues for non-urgent predictions

📊 Monitoring & Maintenance

  • Weekly Evaluation: Compare predictions with actual churn
  • Quarterly Retraining: Retrain model with latest 12 months of data
  • A/B Testing: Test model updates before full deployment
  • Feature Drift Detection: Monitor if feature distributions change
  • Feedback Loop: Incorporate business feedback into improvements

⚠️ Common Pitfalls to Avoid

  • Training-Serving Skew: Ensure preprocessing is identical in training and production
  • Data Leakage: Don't use future information to predict the past
  • Ignoring Model Decay: Model performance degrades over time - monitor actively
  • Over-automation: Keep humans in the loop for high-stakes decisions
  • Poor Error Handling: Gracefully handle missing data and edge cases
